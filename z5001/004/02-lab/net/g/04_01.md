That's a good design for demonstrating **producer-consumer** and **thread synchronization**.

Instead of every client thread calling `input()`, let them **queue incoming messages**. The **Admin thread** is the only one that reads the keyboard, picks a queued message, and sends the reply.

### Architecture

```text
                +-------------------+
                |   Admin Thread    |
                |-------------------|
                | input()           |
                | Reply to client   |
                +---------+---------+
                          ^
                          |
                  Message Queue
                          ^
                          |
        +-----------------+------------------+
        |                 |                  |
+---------------+ +---------------+ +---------------+
| Client A      | | Client B      | | Client C      |
| recv()        | | recv()        | | recv()        |
| enqueue()     | | enqueue()     | | enqueue()     |
+---------------+ +---------------+ +---------------+
```

## Server

```python
import socket
import threading
import queue

HOST = "127.0.0.1"
PORT = 5000

clients = {}
message_queue = queue.Queue()
running = True


def handle_client(client, address):
    print(f"{address} connected")
    clients[address] = client

    try:
        while running:
            data = client.recv(1024)

            if not data:
                break

            msg = data.decode()

            if msg.upper() == "END":
                message_queue.put((address, client, "Client ended conversation"))
                break

            message_queue.put((address, client, msg))

    except:
        pass

    clients.pop(address, None)
    client.close()
    print(f"{address} disconnected")


def admin_console():
    global running

    while running:

        # wait until a client sends something
        address, client, msg = message_queue.get()

        print("\n----------------------------")
        print(f"From {address}")
        print(f"Message : {msg}")

        reply = input("Reply (END/SHUTDOWN/text): ")

        if reply.upper() == "SHUTDOWN":

            running = False

            for c in list(clients.values()):
                try:
                    c.send("SERVER_SHUTDOWN".encode())
                    c.close()
                except:
                    pass

            server.close()
            print("Server stopped.")
            break

        client.send(reply.encode())

        if reply.upper() == "END":
            client.close()


server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind((HOST, PORT))
server.listen()

threading.Thread(target=admin_console, daemon=True).start()

print("Server started...")

while running:
    try:
        client, address = server.accept()

        threading.Thread(
            target=handle_client,
            args=(client, address),
            daemon=True
        ).start()

    except:
        break
```

## Client

```python
import socket

HOST = "127.0.0.1"
PORT = 5000

client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
client.connect((HOST, PORT))

print("Connected")

while True:

    msg = input("You : ")

    client.send(msg.encode())

    if msg.upper() == "END":
        break

    reply = client.recv(1024).decode()

    if reply == "END":
        print("Server ended conversation.")
        break

    if reply == "SERVER_SHUTDOWN":
        print("Server shutting down.")
        break

    print("Server:", reply)

client.close()
print("Disconnected")
```

### Why this is good for teaching

* Only **one `input()`** exists in the entire server.
* Client threads **only receive data** and place it into a shared `Queue`.
* The admin thread **consumes** queued messages and decides the response.
* It naturally introduces:
  * `threading.Thread`
  * `queue.Queue`
  * producer–consumer pattern
  * synchronization without explicit locks
