## Pseudocode
```
FUNCTION findKthSmallest(m, n, k)

    low ← 1
    high ← m × n

    WHILE low < high

        mid ← low + (high - low) / 2

        count ← 0

        FOR i ← 1 TO m

            count ← count + MIN(mid / i, n)

        END FOR

        IF count < k

            low ← mid + 1

        ELSE

            high ← mid

        END IF

    END WHILE

    RETURN low

END FUNCTION
```
