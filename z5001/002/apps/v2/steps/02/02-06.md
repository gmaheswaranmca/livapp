# Code

```jsx
import Editor from "@monaco-editor/react";

function MonacoEditorWrapper({

  language,

  theme,

  fontSize,

  wordWrap,

  code,

  setCode,

}) {

  return (

    <div className="h-full w-full">

      <Editor

        height="100%"

        language={language}

        theme={theme}

        value={code}

        onChange={(value) =>

          setCode(value || "")

        }

        options={{

          fontSize,

          tabSize: 4,

          automaticLayout: true,

          wordWrap,

          minimap: {

            enabled: true,

          },

          scrollBeyondLastLine: false,

          smoothScrolling: true,

          fontLigatures: true,

        }}

      />

    </div>

  );

}

export default MonacoEditorWrapper;
```

# Better Approach

Create:

```text id="mbp4t0"
utils/getMonacoLanguage.js
```

Example:

```javascript
export function getMonacoLanguage(lang){

  const map = {

    python:"python",

    javascript:"javascript",

    cpp:"cpp",

    r:"r"

  };

  return map[lang] || "plaintext";

}
```

Then:

```jsx
language={getMonacoLanguage(language)}
```

This prevents future issues.
