
Code:

```jsx
function TabBar({

  activeTab,

  setActiveTab

}) {

  return (

    <div

      className="

      flex

      border-b

      "

    >

      {

      ["testcases","output"]

      .map(tab=>(

        <button

          key={tab}

          onClick={()=>

            setActiveTab(tab)

          }

          className={`

            px-5

            py-2

            border-r

            ${

            activeTab===tab

            ?

            "bg-gray-200"

            :

            "bg-white"

            }

          `}

        >

          {tab}

        </button>

      ))

      }

    </div>

  );

}

export default TabBar;
```

# 4. BottomPanel.jsx

File:

```text
src/components/editor/BottomPanel.jsx
```

Code:

```jsx
import TabBar from "./TabBar";

import OutputPanel from "./OutputPanel";

import TestCasePanel

from "./TestCasePanel";


function BottomPanel({

  activeTab,

  setActiveTab,

  output

}) {

  return (

    <div

    className="

    h-full

    flex

    flex-col

    border-t

    "

    >

      <TabBar

       activeTab={activeTab}

       setActiveTab={setActiveTab}

      />



      <div className="flex-1">

      {

      activeTab==="output"

      ?

      (

      <OutputPanel

       output={output}

      />

      )

      :

      (

      <TestCasePanel/>

      )

      }

      </div>

    </div>

  );

}

export default BottomPanel;
```
