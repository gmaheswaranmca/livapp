### AppLangEditor.jsx

```jsx
import { useState } from "react";

import Toolbar from "./Toolbar";

import ResizableEditorLayout
from "../layout/ResizableEditorLayout";

export default function AppLangEditor() {

  const [running,setRunning] = useState(false);

  const [output,setOutput] = useState("");



  const runCode = async()=>{

      setRunning(true);

      setOutput(
          "Runtime will be connected in Phase 3"
      );

      setTimeout(()=>{

          setRunning(false);

      },1000)

  }



  const resetCode=()=>{

      window.location.reload();

  }



  return(

      <div className="h-screen flex flex-col">

          <Toolbar

              onRun={runCode}

              onReset={resetCode}

              running={running}

          />


          <div className="flex-1 overflow-hidden">

              <ResizableEditorLayout

                  output={output}

              />

          </div>

      </div>

  )

}
```

# 5. storage.js

This utility will save:

* Selected language
* Theme
* Font
* Editor code

Path

```text
src/utils/storage.js
```

```javascript
export const save = (key,value)=>{

    localStorage.setItem(

        key,

        JSON.stringify(value)

    );

}



export const load=(key)=>{

    const value=localStorage.getItem(key);

    return value

        ? JSON.parse(value)

        : null;

}
```
