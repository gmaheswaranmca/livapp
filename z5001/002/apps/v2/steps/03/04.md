# Complete WebRRuntime.js

```javascript
import BaseRuntime from "../BaseRuntime";

import { WebR } from "webr";


export default class WebRRuntime
extends BaseRuntime{


    constructor(){

        super();

        this.webR=null;

        this.initialized=false;

    }



    async initialize(){

        if(this.initialized){

            return;

        }


        this.webR=new WebR();


        await this.webR.init();


        this.initialized=true;

    }




    async execute(code){

        try{

            const output=

            await this.webR.captureR(code);


            return{

                output:

                output.output

                ||

                JSON.stringify(output),

                error:false

            };

        }

        catch(err){

            return{

                output:

                err.toString(),

                error:true

            };

        }

    }

}
```
