# Final PyodideRuntime.js

```javascript
import BaseRuntime from "../BaseRuntime";

export default class PyodideRuntime
extends BaseRuntime {

    constructor(){

        super();

        this.pyodide=null;

        this.initialized=false;

    }


    async initialize(){

        if(this.initialized){

            return;

        }


        const {loadPyodide}

        = await import(

        "https://cdn.jsdelivr.net/pyodide/v0.28.3/full/pyodide.mjs"

        );


        this.pyodide=

        await loadPyodide({

        indexURL:

        "https://cdn.jsdelivr.net/pyodide/v0.28.3/full/"

        });


        this.initialized=true;

    }



    async execute(code){

        let output=[];


        this.pyodide.setStdout({

            batched:(msg)=>{

                output.push(msg);

            }

        });


        this.pyodide.setStderr({

            batched:(msg)=>{

                output.push(msg);

            }

        });



        try{


            await this.pyodide

            .runPythonAsync(code);


            return{

                output:

                output.join("\n"),

                error:false

            };

        }

        catch(err){

            return{

                output:

                err.toString(),

                error:true

            }

        }

    }

}
```
