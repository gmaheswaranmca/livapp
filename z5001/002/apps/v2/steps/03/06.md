## RuntimeManager.js

```javascript
import RuntimeFactory from "./RuntimeFactory";

class RuntimeManager {

    constructor(){

        this.cache = {};

    }


    async get(language){

        if(this.cache[language]){

            return this.cache[language];

        }


        const runtime = await RuntimeFactory.create(language);

        await runtime.initialize();

        this.cache[language] = runtime;

        return runtime;

    }


    async destroy(language){

        const runtime=this.cache[language];

        if(!runtime){

            return;

        }

        if(runtime.destroy){

            await runtime.destroy();

        }

        delete this.cache[language];

    }


    async destroyAll(){

        for(const lang in this.cache){

            const runtime=this.cache[lang];

            if(runtime.destroy){

                await runtime.destroy();

            }

        }

        this.cache={};

    }

}


export default new RuntimeManager();
```

# Improve JS Runtime

Current:

```javascript
eval(code)
```

This is risky.

Prefer:

```javascript
new Function(code)();
```

Example:

```javascript
async execute(code){

    const logs=[];

    const original=console.log;


    console.log=(...args)=>{

        logs.push(args.join(" "));

    }


    try{

        const fn=new Function(code);

        fn();

    }

    catch(err){

        logs.push(err.toString());

    }


    console.log=original;


    return{

        output:logs.join("\n"),

        error:false

    };

}
```
