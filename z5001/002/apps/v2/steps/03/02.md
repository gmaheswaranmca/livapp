Path


---

### Version 1

```javascript
import BaseRuntime from "../BaseRuntime";

export default class JSRuntime
extends BaseRuntime {


    async initialize(){

        return true;

    }



    async execute(code){

        const logs=[];

        const original=console.log;


        console.log=(...args)=>{

            logs.push(

                args.join(" ")

            );

        }


        try{

            eval(code);

        }

        catch(err){

            logs.push(

                err.toString()

            );

        }


        console.log=original;


        return {

            output:logs.join("\n"),

            error:false

        }

    }


}
```

# Step 3 : RuntimeFactory.js

We don't want

```javascript
if(lang==="python")

if(lang==="r")

if(lang==="cpp")
```

all over the app.

So:

```javascript
import JSRuntime from "./javascript/JSRuntime";

export default class RuntimeFactory {

    static async create(language){

        switch(language){

            case "javascript":

                return new JSRuntime();


            default:

                throw new Error(

                    "Runtime unavailable"

                );

        }

    }

}
```

---

Replace runCode

```javascript
const runCode=async()=>{

    setRunning(true);

    try{

        const runtime=

            await RuntimeFactory

            .create(language);


        await runtime.initialize();


        const result=

            await runtime.execute(code);


        setOutput(

            result.output

        );

    }

    catch(err){

        setOutput(

            err.toString()

        );

    }


    setRunning(false);

}
```
