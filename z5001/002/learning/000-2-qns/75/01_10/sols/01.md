        counter = 0

        res = []

class Solution:
    def mergeAlternately(self, word1: str, word2: str) -> str:
        if not word1:
            return word2
        if not word2:
            return word1

        merge = [] 
        #list, the merge all the chars at the end
        # track length of each word, if we are at 
        # the end of one word, we append the other 
        # since both not empty
        # 3 cases: either both in range, 1 is empty but 2 is not, vice versa

        #single pointer i: if i is in range for both word1 and word2,
        # then add word1[i] first then word2[i] to merge. 
        # if either out of range, only add the other
        max_len = max(len(word1),len(word2))
        for i in range(max_len): 
            #we append this many chars into merge
            if i<len(word1) and i<len(word2): 
                merge.append(word1[i])
                merge.append(word2[i])
            elif i>=len(word1) and i<len(word2):
                #then i is pointing out of range but j is in range
                merge.append(word2[i])
            elif i<len(word1) and i>=len(word2):
                merge.append(word1[i])

        return "".join(merge)
