✅ *Q1. Climbing Stairs*  
*Problem:* Count ways to climb `n` stairs (1 or 2 steps).  
```python
def climbStairs(n):
    if n <= 2: return n
    a, b = 1, 2
    for _ in range(3, n+1):
        a, b = b, a + b
    return b
```

✅ *Q2. Coin Change*  
*Problem:* Minimum coins to make amount using given coins.  
```python
def coinChange(coins, amount):
    dp = [float('inf')] * (amount+1)
    dp = 0
    for i in range(1, amount+1):
        for c in coins:
            if i - c >= 0:
                dp[i] = min(dp[i], dp[i-c] + 1)
    return dp[amount] if dp[amount]!= float('inf') else -1
```

✅ *Q3. Longest Increasing Subsequence*  
*Problem:* Find length of LIS in array.  
```python
def lengthOfLIS(nums):
    dp =*len(nums)
    for i in range(len(nums)):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp) if dp else 0
```

✅ *Q4. 0/1 Knapsack*  
*Problem:* Max value with given weights and capacity.  
```python
def knapsack(weights, values, W):
    dp =*(W+1)
    for i in range(len(weights)):
        for w in range(W, weights[i]-1, -1):
            dp[w] = max(dp[w], dp[w-weights[i]]+values[i])
    return dp[W]
```

✅ *Q5. Edit Distance*  
*Problem:* Min operations to convert one string to another.  
```python
def minDistance(word1, word2):
    m, n = len(word1), len(word2)
    dp = [*(n+1) for _ in range(m+1)]
    for i in range(m+1): dp[i] = i
    for j in range(n+1): dp[j] = j
    for i in range(1, m+1):
        for j in range(1, n+1):
            if word1[i-1] == word2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]
```
